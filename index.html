import sqlite3 import json def export_collection_to_json(sport_filter=None, output_filename="collection_data.json"): conn = sqlite3.connect('collector_workbench.db') # Row factory allows us to easily convert rows into dictionaries conn.row_factory = sqlite3.Row cursor = conn.cursor() if sport_filter: # Export a specific subset (e.g., just 'Soccer' or 'Rugby') cursor.execute(''' SELECT id, sport, player_name, year, brand, card_number, status, date_sent, date_received, notes FROM collection WHERE sport COLLATE NOCASE = ? ''', (sport_filter,)) print(f"Exporting subset for sport: {sport_filter}...") else: # Export everything across all sports cursor.execute(''' SELECT id, sport, player_name, year, brand, card_number, status, date_sent, date_received, notes FROM collection ''') print("Exporting entire multi-sport collection...") rows = cursor.fetchall() conn.close() # Convert database rows into a standard list of dictionaries collection_list = [dict(row) for row in rows] # Write to a JSON file formatted nicely for the web with open(output_filename, 'w', encoding='utf-8') as f: json.dump(collection_list, f, indent=4) print(f"Successfully exported {len(collection_list)} record(s) to '{output_filename}'!") if __name__ == "__main__": # Example usage: # Export just your soccer cards for your webpage: # export_collection_to_json(sport_filter="Soccer", output_filename="soccer_cards.json") # Or export the entire collection: # export_collection_to_json(output_filename="all_sports.json")

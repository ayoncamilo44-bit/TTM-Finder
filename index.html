import sqlite3 import csv # 1. Initialize the Database def init_workbench_db(): conn = sqlite3.connect('collector_workbench.db') cursor = conn.cursor() cursor.execute(''' CREATE TABLE IF NOT EXISTS collection ( id INTEGER PRIMARY KEY AUTOINCREMENT, player_name TEXT NOT NULL, year INTEGER, brand TEXT, card_number TEXT, status TEXT, -- e.g., 'Owned', 'Sent TTM', 'Received', 'Wishlist' date_sent TEXT, date_received TEXT, notes TEXT ) ''') conn.commit() conn.close() print("Database initialized successfully!") # 2. Add a Card or TTM Request def add_card(player_name, year, brand, card_number, status, date_sent=None, notes=""): conn = sqlite3.connect('collector_workbench.db') cursor = conn.cursor() cursor.execute(''' INSERT INTO collection (player_name, year, brand, card_number, status, date_sent, notes) VALUES (?, ?, ?, ?, ?, ?, ?) ''', (player_name, year, brand, card_number, status, date_sent, notes)) conn.commit() conn.close() print(f"Added: {year} {brand} {player_name} ({status})") # 3. Query Pending TTM Requests def get_pending_ttm(): conn = sqlite3.connect('collector_workbench.db') cursor = conn.cursor() cursor.execute("SELECT id, player_name, date_sent FROM collection WHERE status = 'Sent TTM'") results = cursor.fetchall() conn.close() return results # 4. Search Collection by Player Name def search_collection(player_query): conn = sqlite3.connect('collector_workbench.db') cursor = conn.cursor() # Using wildcard '%' so partial names work (e.g., "Ryan" matches "Nolan Ryan") cursor.execute(''' SELECT id, year, brand, card_number, status, date_sent, notes FROM collection WHERE player_name LIKE ? ''', (f'%{player_query}%',)) results = cursor.fetchall() conn.close() if results: print(f"\nFound {len(results)} result(s) for '{player_query}':") for row in results: print(f"ID: {row[0]} | {row[1]} {row[2]} #{row[3]} | Status: {row[4]} | Notes: {row[6]}") else: print(f"\nNo cards found for player matching '{player_query}'.") return results # 5. Import from CSV def import_csv_to_workbench(csv_filename): conn = sqlite3.connect('collector_workbench.db') cursor = conn.cursor() with open(csv_filename, mode='r', encoding='utf-8') as file: reader = csv.DictReader(file) for row in reader: cursor.execute(''' INSERT INTO collection (player_name, year, brand, card_number, status, notes) VALUES (?, ?, ?, ?, ?, ?) ''', ( row.get('Player'), row.get('Year'), row.get('Brand'), row.get('Card Number'), row.get('Status', 'Owned'), row.get('Notes', '') )) conn.commit() conn.close() print(f"Successfully imported data from {csv_filename}!") if __name__ == "__main__": init_workbench_db() # Example test search after running: # search_collection("Ryan")
